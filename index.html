Materi Stack

Stack (tumpukan) adalah kumpulan data yang diletakkan di atas data lain dengan konsep LIFO (Last In, First Out): elemen terakhir yang disimpan menjadi elemen pertama yang diambil. Meletakkan elemen di atas disebut push, mengambilnya disebut pop.

Sumber: Modul Praktikum Algoritma dan Struktur Data, Modul IV — Stack (tujuan, dasar teori, operasi pada stack, latihan troubleshooting).

Representasi: struct dengan dua field

Stack disajikan memakai array dengan anggapan jumlah elemen maksimum tidak melebihi ukuran array. Agar overflow bisa dideteksi, dibutuhkan data tambahan berupa posisi ujung stack, sehingga dipakai struct dengan dua field: array untuk menyimpan elemen dan integer top untuk mencatat posisi ujung.

struct STACK {          // jenis data abstrak 'STACK'
    int   top;          // posisi ujung stack (-1 = kosong)
    float data[maxstack];
};
float dta;
struct STACK stackbaru;

Operasi dasar (dari modul)

bool isempty() {   // kosongkah?
    if (stackbaru.top == -1) return true;
    else return false;
}

bool isfull() {    // penuhkah?
    if (stackbaru.top == maxstack - 1)
        return true;
    else return false;
}
void clear() {     // hapus seluruh stack
    stackbaru.top = -1;
}

void print() {     // cetak isi stack
    for (int i = 0; i <= stackbaru.top; i++)
        cout << stackbaru.data[i] << "  ";
}
  • isempty: stack kosong jika top == -1.
  • isfull: stack penuh jika top == maxstack - 1 (index terakhir array).
  • clear: cukup mengembalikan top ke -1, data lama dianggap tidak ada.
  • print: membaca dari index 0 sampai top.

Push & Pop yang benar

void push(float dta) {
    if (isfull()) puts("Maaf, stack penuh");
    else {
        stackbaru.top++;                       // naikkan top dulu
        stackbaru.data[stackbaru.top] = dta;   // lalu simpan
    }
}
void pop() {
    if (isempty()) cout << "Data telah kosong!";
    else {
        cout << "data yang terambil : " << stackbaru.data[stackbaru.top] << endl;
        stackbaru.top--;                       // ambil dulu, lalu turunkan top
    }
}

Mengapa stack cocok?

Push memasukkan karakter dari kiri ke kanan; pop mengeluarkannya dari yang terakhir masuk. Urutan keluar adalah kebalikan urutan masuk (LIFO). Waktu O(n) (n kali push, n kali pop) dan memori tambahan O(n) untuk stack.

Implementasi dengan array: struct

Queue disimpan dalam struct berisi array elemen, penanda first (depan), last (belakang), dan number_of_items. Catatan: modul tidak menuliskan typedef int Titem, tetapi tipe ini perlu ada agar program dapat dikompilasi.

#define MAXN 6
typedef int Titem;
typedef enum {NOT_OK, OK} Tboolean;
typedef struct {
    Titem array[MAXN];
    int first;            // posisi elemen pertama (depan)
    int last;             // posisi elemen terakhir (belakang)
    int number_of_items;  // jumlah elemen saat ini
} Tqueue;

Enqueue & dequeue (array melingkar)

Awalnya first = 0, last = -1, number_of_items = 0. Saat last atau first melewati index terakhir, keduanya kembali ke 0 (wrap), sehingga slot kosong di depan dapat dipakai lagi. Penuh atau kosong ditentukan oleh number_of_items, bukan oleh posisi first/last.

bool enqueue(Tqueue *Pqueue, Titem item) {
  if (Pqueue->number_of_items >= MAXN)
    return NOT_OK;          // penuh
  else {
    Pqueue->last++;
    if (Pqueue->last > MAXN-1)
      Pqueue->last = 0;     // wrap
    Pqueue->array[Pqueue->last] = item;
    Pqueue->number_of_items++;
    return OK;
  }
}
bool dequeue(Tqueue *Pqueue, Titem *Pitem) {
  if (Pqueue->number_of_items == 0)
    return NOT_OK;          // kosong
  else {
    *Pitem = Pqueue->array[Pqueue->first++];
    if (Pqueue->first > MAXN-1)
      Pqueue->first = 0;    // wrap
    Pqueue->number_of_items--;
    return OK;
  }
}

Implementasi dengan pointer (linked list)

Setiap elemen adalah node dengan nextPtr. headPtr menunjuk elemen depan, tailPtr elemen belakang. Enqueue menyambung node baru di belakang; dequeue mengambil node di depan. Kapasitas tidak dibatasi array.

struct queueNode { char data; struct queueNode *nextPtr; };
typedef struct queueNode QUEUENODE;
typedef QUEUENODE *QUEUENODEPTR;
QUEUENODEPTR headPtr = NULL, tailPtr = NULL;

int isEmpty(QUEUENODEPTR headPtr) { return headPtr == NULL; }

void enqueue(QUEUENODEPTR *headPtr, QUEUENODEPTR *tailPtr, char value) {
    QUEUENODEPTR newPtr = malloc(sizeof(QUEUENODE));
    if (newPtr != NULL) {
        newPtr->data = value;
        newPtr->nextPtr = NULL;
        if (isEmpty(*headPtr)) *headPtr = newPtr;   // queue masih kosong
        else (*tailPtr)->nextPtr = newPtr;          // sambung di belakang
        *tailPtr = newPtr;
    }
}
// dequeue (pelengkap): ambil dari depan
char dequeue(QUEUENODEPTR *headPtr, QUEUENODEPTR *tailPtr) {
    QUEUENODEPTR tempPtr = *headPtr;
    char value = (*headPtr)->data;
    *headPtr = (*headPtr)->nextPtr;
    if (*headPtr == NULL) *tailPtr = NULL;   // queue jadi kosong
    free(tempPtr);
    return value;
}

Latihan modul: queue linear dengan insert dan del

Latihan pertama memakai array biasa dengan front = rear = -1. rear naik tiap insert, front naik tiap delete, dan queue kosong bila front == rear. Kelemahannya: front dan rear tidak pernah kembali, sehingga setelah MAX insert queue dianggap penuh walau semua elemen sudah dihapus. Masalah ini diselesaikan oleh array melingkar pada tab Simulasi.

void insert(int queue[], int *rear, int nilai) {
    if (*rear < MAX-1) { *rear = *rear + 1; queue[*rear] = nilai; }
    else { cout << "queue penuh, insert tidak dapat dilakukan" << endl; exit(0); }
}
void del(int queue[], int *front, int rear, int *nilai) {
    if (*front == rear) { cout << "queue kosong, delete tidak dapat dilakukan" << endl; exit(0); }
    *front = *front + 1;          // front menunjuk satu posisi sebelum elemen depan
    *nilai = queue[*front];
}
// main: front = rear = -1  (kosong jika front == rear)
